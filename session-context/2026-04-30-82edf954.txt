```python
from flask import Flask, request
import requests
import os

app = Flask(__name__)

@app.route('/email', methods=['POST'])
def receive_email():
    data = request.json
    email_body = data['message']['text']
    sender = data['from']['email']
    
    # Only process if email contains "CUSTOMER"
    if "CUSTOMER" not in email_body.upper():
        return {"status": "ignored - no CUSTOMER keyword"}
    
    response_text = generate_response(email_body)
    
    # Only text if email also contains "INVOICE"
    if "INVOICE" in email_body.upper():
        send_sms("+1234567890", f"Response generated: {response_text[:50]}...")
    
    return {"status": "ok"}

def generate_response(email_body):
    response = requests.post("https://api.openai.com/v1/chat/completions", 
        json={"model": "gpt-4o-mini", "messages": [{"role": "user", "content": email_body}]},
        headers={"Authorization": f"Bearer {os.getenv('OPENAI_API_KEY')}"}
    )
    return response.json()['choices'][0]['message']['content']

def send_sms(phone, message):
    requests.post(f"https://api.telnyx.com/v2/messages",
        json={"to": phone, "from": os.getenv('TELNYX_FROM'), "text": message},
        headers={"Authorization": f"Bearer {os.getenv('TELNYX_API_KEY')}"}
    )

if __name__ == '__main__':
    app.run()
```

**Changes:**
- Email skipped entirely if missing "CUSTOMER"
- Response only generated if "CUSTOMER" exists
- SMS only sent if both "CUSTOMER" AND "INVOICE" are present
